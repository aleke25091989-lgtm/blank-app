<script>
    function tog(el) { 
        const c = el.nextElementSibling; 
        const isVisible = c.style.display === "block";
        c.style.display = isVisible ? "none" : "block"; 
        el.querySelector('span').innerText = isVisible ? "+" : "-";
    }

    function show(id) { 
        document.querySelectorAll('.med-section').forEach(s => s.style.display = 'none'); 
        document.getElementById(id).style.display = 'block'; 
        window.scrollTo(0, 150);
    }

    function search(q) {
        q = q.toLowerCase();
        document.querySelectorAll('.med-card').forEach(m => {
            m.style.display = m.innerText.toLowerCase().includes(q) ? "block" : "none";
        });
        if(q.length > 0) document.querySelectorAll('.med-section').forEach(s => s.style.display = 'block');
    }

    function res(id, val) { document.getElementById(id).innerText = val.toFixed(2) + " ml/h"; }

    function cal_nora() {
        let w = document.getElementById('w_nora').value;
        let d = document.getElementById('d_nora').value;
        let c = document.getElementById('c_nora').value;
        let v = (d * w * 60) / (c * 1000);
        let out = v.toFixed(2) + " ml/h";
        if(v > 5 && c == 0.5) out += "<br><small style='color:#ff3e3e'>⚠️ PASSER EN 1 mg/ml</small>";
        document.getElementById('res_nora').innerHTML = out;
    }

    function cal_dobu() {
        let w = document.getElementById('w_dobu').value;
        let mg = w * 3;
        let ml_amp = (mg * 20) / 250;
        document.getElementById('res_dobu').innerHTML = `Prélèvement : ${ml_amp.toFixed(1)}ml de produit + G5% QSP 50ml`;
    }

    // ECG Animation
    const canvas = document.getElementById('ecg-bg');
    const ctx = canvas.getContext('2d');
    let x = 0;
    function resize() { canvas.width = window.innerWidth; canvas.height = window.innerHeight; }
    window.onresize = resize; resize();
    function draw() {
        ctx.fillStyle = "rgba(10, 47, 95, 0.05)";
        ctx.fillRect(0, 0, canvas.width, canvas.height);
        ctx.strokeStyle = "#00e5ff";
        ctx.lineWidth = 1.5;
        ctx.beginPath(); ctx.moveTo(x, canvas.height/2);
        x += 2.5;
        let y = canvas.height/2;
        if(x % 150 < 4) y -= 40; if(x % 150 >= 4 && x % 150 < 8) y += 50;
        ctx.lineTo(x, y); ctx.stroke();
        if (x > canvas.width) x = 0;
        requestAnimationFrame(draw);
    }
    draw();
</script>
